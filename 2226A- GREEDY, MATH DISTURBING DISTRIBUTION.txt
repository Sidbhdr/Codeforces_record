import java.util.*;
import java.io.*;

public class Main {
    static final long MOD = 676767677L;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringBuilder out = new StringBuilder();
        int t = Integer.parseInt(br.readLine().trim());
        while (t-- > 0) {
            int n = Integer.parseInt(br.readLine().trim());
            String[] parts = br.readLine().trim().split(" ");
            int[] a = new int[n];
            for (int i = 0; i < n; i++) a[i] = Integer.parseInt(parts[i]);

            // dp[i] = minimum cost to remove first i elements
            long[] dp = new long[n + 1];
            Arrays.fill(dp, Long.MAX_VALUE / 2);
            dp[0] = 0;

            for (int i = 0; i < n; i++) {
                // try subsequence starting at i
                long product = 1;
                for (int j = i; j < n; j++) {
                    if (j > i && a[j] < a[j - 1]) break; // must be non-decreasing
                    product = (product * a[j]) % MOD;
                    dp[j + 1] = Math.min(dp[j + 1], (dp[i] + product) % MOD);
                }
            }

            out.append(dp[n] % MOD).append("\n");
        }
        System.out.print(out);
    }
}
