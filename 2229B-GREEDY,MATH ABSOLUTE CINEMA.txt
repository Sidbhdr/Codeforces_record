import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int t = sc.nextInt();

        while (t-- > 0) {
            int n = sc.nextInt();

            long[] a = new long[n];
            long[] b = new long[n];

            for (int i = 0; i < n; i++) {
                a[i] = sc.nextLong();
            }

            for (int i = 0; i < n; i++) {
                b[i] = sc.nextLong();
            }

            long sumLarge = 0;
            long maxSmall = 0;

            for (int i = 0; i < n; i++) {
                sumLarge += Math.max(a[i], b[i]);
                maxSmall = Math.max(maxSmall, Math.min(a[i], b[i]));
            }

            System.out.println(sumLarge + maxSmall);
        }

        sc.close();
    }
}
